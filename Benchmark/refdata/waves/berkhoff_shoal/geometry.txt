# Berkhoff, Booy & Radder (1982) elliptic shoal on a 1:50 slope (Coastal Engineering 6, 255-279)

# [S1] Basilisk example https://basilisk.fr/src/examples/shoal.c (raw ?raw), code verbatim:
#   double h0 = 0.45;
#   double cosa = cos (20.*pi/180.), sina = sin (20.*pi/180.);
#   double xr = x*cosa - y*sina, yr = x*sina + y*cosa;
#   double z0 = xr >= -5.82 ? (5.82 + xr)/50. : 0.;
#   double zs = sq(xr/3.) + sq(yr/4.) <= 1. ? -0.3 + 0.5*sqrt(1. - sq(xr/3.75) - sq(yr/5.)) : 0.;
#   zb[] = z0 + zs - h0;
#   i.e. depth h = 0.45 for x' < -5.82 ; h = 0.45 - 0.02*(5.82 + x') otherwise (slope 1:50, contours rotated 20 deg),
#   minus the shoal thickness -0.3 + 0.5*sqrt(1 - (x'/3.75)^2 - (y'/5)^2) inside (x'/3)^2 + (y'/4)^2 <= 1.
#   Basilisk domain X0 = -10, L0 = 25, Y0 = -L0/2 ; T0 = 1 s.
#   Section lines (gnuplot labels/interpolation in shoal.c): section 2 at x = 3, section 3 at x = 5,
#   section 5 at x = 9, section 7 along y = 0 [m]; model normalised by a0 = 0.026 m.
